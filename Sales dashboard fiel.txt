Use code with caution.Open backend/main.py and paste this exact python script:pythonimport os, json, time, gspread, pandas as pd
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from google.oauth2.service_account import Credentials

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"])

CACHED_DATA, LAST_FETCH_TIME, CACHE_TTL = None, 0, 3600

@app.get("/api/data")
def get_dashboard_data():
    global CACHED_DATA, LAST_FETCH_TIME
    current_time = time.time()
    if CACHED_DATA is not None and (current_time - LAST_FETCH_TIME) < CACHE_TTL:
        return {"status": "success", "data": CACHED_DATA}

    scopes = ["https://googleapis.com", "https://googleapis.com"]
    creds = Credentials.from_service_account_info(json.loads(os.environ.get("GOOGLE_CREDENTIALS")), scopes=scopes)
    sheet = gspread.authorize(creds).open_by_url(os.environ.get("GOOGLE_SHEET_URL")).sheet1

    CACHED_DATA = pd.DataFrame(sheet.get_all_records()).to_dict(orient="records")
    LAST_FETCH_TIME = current_time
    return {"status": "success", "data": CACHED_DATA}
